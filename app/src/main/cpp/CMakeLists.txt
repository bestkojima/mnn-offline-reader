cmake_minimum_required(VERSION 3.22)
project(english_reader LANGUAGES C CXX ASM)
if(ANDROID)
    # Keep inference optimized in the debug APK too; Java remains debuggable.
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Native inference build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_JOB_POOLS reader_compile=4)
set(CMAKE_JOB_POOL_COMPILE reader_compile)
get_filename_component(READER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(MNN_ROOT "${READER_ROOT}/vendor/MNN" CACHE PATH "Pinned MNN source checkout")
if(NOT EXISTS "${MNN_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "MNN source missing. Run git submodule update --init --depth 1, or set MNN_ROOT.")
endif()
set(MNN_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_CONVERTER OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(MNN_INTERNAL OFF CACHE BOOL "" FORCE)
set(MNN_LOW_MEMORY ON CACHE BOOL "" FORCE)
set(MNN_ARM82 ON CACHE BOOL "" FORCE)
set(MNN_SME2 OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "" FORCE)
add_subdirectory("${MNN_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/mnn")
add_library(reader_engine STATIC supertonic_engine.cpp kokoro_engine.cpp)
target_include_directories(reader_engine PUBLIC "${MNN_ROOT}/include" "${CMAKE_CURRENT_LIST_DIR}")
target_compile_options(reader_engine PRIVATE -fno-exceptions -fno-rtti)
target_link_libraries(reader_engine PUBLIC MNN)
if(ANDROID)
    add_library(english_reader SHARED reader_jni.cpp)
    target_compile_options(english_reader PRIVATE -fno-exceptions -fno-rtti)
    target_link_options(english_reader PRIVATE "-Wl,-z,max-page-size=16384")
    target_link_libraries(english_reader PRIVATE reader_engine log)
else()
    add_executable(reader_smoke smoke_main.cpp)
    target_include_directories(reader_smoke PRIVATE "${MNN_ROOT}/apps/frameworks/3rd_party/include")
    target_compile_definitions(reader_smoke PRIVATE JSON_NOEXCEPTION=1)
    target_compile_options(reader_smoke PRIVATE -fno-exceptions -fno-rtti)
    target_link_libraries(reader_smoke PRIVATE reader_engine)
    add_executable(kokoro_smoke kokoro_smoke.cpp)
    target_compile_options(kokoro_smoke PRIVATE -fno-exceptions -fno-rtti)
    target_link_libraries(kokoro_smoke PRIVATE reader_engine)
endif()
